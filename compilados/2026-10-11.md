def search_europe_pmc(query, limit=5):
    """Busca papers en Europe PMC API"""
    encoded_query = urllib.parse.quote(query)
    url = f"https://www.ebi.ac.uk/europepmc/webservices/rest/search?query={encoded_query}&format=json&resultType=core&pageSize={limit}"
    
    req = urllib.request.Request(url, headers={
        'User-Agent': 'ConsciousnessResearchBot/2.0 (mailto:research@example.com)'
    })
    
    try:
        with urllib.request.urlopen(req, timeout=30) as response:
            data = json.loads(response.read().decode())
            results = data.get('resultList', {}).get('result', [])
            
            papers = []
            for r in results:
                paper = {
                    'title': r.get('title', 'Sin título').strip(),
                    'authors': r.get('authorString', 'Autores no disponibles'),
                    'journal': r.get('journalTitle', 'Revista no especificada'),
                    'year': r.get('pubYear', 'N/A'),
                    'doi': r.get('doi', ''),
                    'pmid': r.get('pmid', ''),
                    'pmcid': r.get('pmcid', ''),
                    'abstract': r.get('abstractText', 'Abstract no disponible'),
                    'url': f"https://doi.org/{r.get('doi', '')}" if r.get('doi') else f"https://europepmc.org/article/{r.get('id', '')}",
                    'pdf_url': f"https://europepmc.org/backend/ptpmcrender.fcgi?accid={r.get('pmcid', '')}&blobtype=pdf" if r.get('pmcid') else '',
                    'is_open_access': r.get('isOpenAccess', False) == 'Y'
                }
                papers.append(paper)
            
            return papers
    except Exception as e:
        print(f"  ⚠ Error buscando '{query[:50]}...': {e}")
        return []

### 1. Fundamentos teóricos y neurobiológicos de GNWT
* **Arquitectura modular vs. global:** Distinguimos los módulos periféricos especializados (inconscientes, de procesamiento local y en paralelo) de la red frontoparietal de axones largos (corteza prefrontal dorsolateral, parietal posterior, cíngulo anterior), que conforma el espacio de trabajo global.

Un estímulo que supera el umbral crítico desencadena un bucle reentrante auto-sostenido acoplado a inhibición lateral (*winner-take-all*), lo que genera el cuello de botella serial de la consciencia.
* **Taxonomía tripartita:** Delimitamos los estados de procesamiento según la fuerza del estímulo y la modulación atencional: **subliminal** (débil, sin ignición), **preconsciente** (fuerte, pero desatendido) y **consciente** (fuerte + atendido = ignición global).
