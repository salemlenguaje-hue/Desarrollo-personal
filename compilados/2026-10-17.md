const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playTone(freq, type, duration, slideTo = null, vol = 0.5) {
    if (!audioCtx || audioCtx.state === 'suspended') return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
    if (slideTo) osc.frequency.exponentialRampToValueAtTime(slideTo, audioCtx.currentTime + duration);
    gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain); 
    gain.connect(masterGain);
    osc.start(); 
    osc.stop(audioCtx.currentTime + duration);
}

let melodyIndex = 0;
function startMusic() {
    if (!musicOn) return;
    const note = melody[melodyIndex];
    if (note[0] > 0) {
        playTone(note[0], 'square', note[1] * 0.9, null, 0.2);
        playTone(note[0] / 2, 'triangle', note[1] * 0.9, null, 0.15);
    }
    melodyIndex = (melodyIndex + 1) % melody.length;
    musicInterval = setTimeout(startMusic, note[1] * 1000);
}

function stopMusic() {
    if (musicInterval) clearTimeout(musicInterval);
}

musicBtn.addEventListener('click', () => {
    musicOn = !musicOn;
    musicBtn.textContent = musicOn ? '🎵 ON' : '🔇 OFF';
    if (musicOn && isPlaying) startMusic();
    else stopMusic();
});

function getScenario() {
    if (score < 500) return 'night';
    if (score < 1000) return 'day';
    return 'sunset';
}

function checkScenarioChange() {
    const newScenario = getScenario();
    if (newScenario !== currentScenario) {
        currentScenario = newScenario;
        if (currentScenario === 'day') scenarioMessage = '☀️ ¡DÍA!';
        else if (currentScenario === 'sunset') scenarioMessage = '🌇 ¡ATARDECER!';
        else scenarioMessage = '🌙 ¡NOCHE!';
        scenarioMessageTimer = 120;
        sfx.scenario();
    }
}

let obstacles = [];
function createObstacle() {
    const rand = Math.random();
    let type;
    if (rand > 0.7) type = 'drone';
    else if (rand > 0.4) type = 'cactus';
    else type = 'bush'; // Nuevo: arbusto bajo

    let h, w, y;
    if (type === 'cactus') { h = 50; w = 30; y = 210; }
    else if (type === 'drone') { h = 30; w = 45; y = 140 + Math.random() * 50; }
    else { h = 25; w = 40; y = 235; } // Bush pegado al suelo

    obstacles.push({ x: canvas.width, y: y, width: w, height: h, type: type, rot: 0 });
}

function drawObstacles() {
    obstacles.forEach(obs => {
        ctx.save();
        const obsColor = currentScenario === 'night' ? '#ff0055' : '#d62828';
        ctx.shadowBlur = 10; ctx.shadowColor = obsColor;

        if (obs.type === 'cactus') {
            ctx.fillStyle = obsColor;
            ctx.beginPath();
            ctx.ellipse(obs.x + 15, obs.y + 25, 8, 25, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.moveTo(obs.x + 8, obs.y + 20);
            ctx.quadraticCurveTo(obs.x - 5, obs.y + 15, obs.x, obs.y + 5);
            ctx.quadraticCurveTo(obs.x + 5, obs.y, obs.x + 8, obs.y + 15);
            ctx.fill();
            ctx.beginPath();
            ctx.moveTo(obs.x + 22, obs.y + 25);
            ctx.quadraticCurveTo(obs.x + 35, obs.y + 20, obs.x + 30, obs.y + 10);
            ctx.quadraticCurveTo(obs.x + 25, obs.y + 5, obs.x + 22, obs.y + 20);
            ctx.fill();
        } else if (obs.type === 'drone') {
            ctx.fillStyle = obsColor;
            ctx.beginPath();
            ctx.ellipse(obs.x + 22, obs.y + 15, 20, 10, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.fillStyle = '#fff';
            ctx.beginPath(); ctx.arc(obs.x + 30, obs.y + 13, 4, 0, Math.PI * 2); ctx.fill();
            ctx.fillStyle = '#000';

function updateObstacles() {
    if (score - lastObstacleScore > 80 + Math.random() * 60) {
        createObstacle();
        lastObstacleScore = score;
    }
    obstacles.forEach((obs, index) => {
        obs.x -= gameSpeed;
        if (obs.x + obs.width < 0) obstacles.splice(index, 1);

        let dTop = dino.ducking ? dino.y + 30 : dino.y;
        let dBot = dino.y + dino.height;
        let dRight = dino.x + (dino.ducking ? dino.width + 10 : dino.width);

        if (dRight > obs.x + 5 && dino.x < obs.x + obs.width - 5 && dBot > obs.y && dTop < obs.y + obs.height) gameOver();
    });
}

let flags = [];
let lastFlagScore = 0;

function createFlag() {
    flags.push({
        x: canvas.width,
        y: 100 + Math.random() * 80,
        width: 45,
        height: 30,
        collected: false,
        wave: 0
    });

function drawFlags() {
    flags.forEach(f => {
        if (f.collected) return;

        ctx.save();
        f.wave = (f.wave + 0.1) % (Math.PI * 2);
        const waveOffset = Math.sin(f.wave) * 2;

        ctx.fillStyle = '#8b4513';
        ctx.fillRect(f.x, f.y - 5, 3, 40);

        const fh = f.height / 3;
        ctx.fillStyle = '#74acdf';
        ctx.fillRect(f.x + 3, f.y + waveOffset, f.width - 3, fh);
        ctx.fillStyle = '#fff';
        ctx.fillRect(f.x + 3, f.y + fh + waveOffset, f.width - 3, fh);
        ctx.fillStyle = '#74acdf';
        ctx.fillRect(f.x + 3, f.y + 2 * fh + waveOffset, f.width - 3, fh);

        ctx.fillStyle = '#fcbf49';
        ctx.shadowBlur = 10; ctx.shadowColor = '#fcbf49';
        const sunX = f.x + 3 + (f.width - 3) / 2;
        const sunY = f.y + f.height / 2 + waveOffset;
        ctx.beginPath();
        ctx.arc(sunX, sunY, 5, 0, Math.PI * 2);
        ctx.fill();

        ctx.strokeStyle = '#fcbf49';
        ctx.lineWidth = 1;
        for (let i = 0; i < 8; i++) {
            const angle = (i / 8) * Math.PI * 2;
            ctx.beginPath();
            ctx.moveTo(sunX + Math.cos(angle) * 6, sunY + Math.sin(angle) * 6);
            ctx.lineTo(sunX + Math.cos(angle) * 9, sunY + Math.sin(angle) * 9);
            ctx.stroke();
        }
        ctx.shadowBlur = 0;

        ctx.restore();
    });
}

function updateFlags() {
    if (score - lastFlagScore > 500 && Math.random() < 0.01) {
        createFlag();
        lastFlagScore = score;
    }

    flags.forEach((f, index) => {
        f.x -= gameSpeed;
        if (f.x + f.width < 0) flags.splice(index, 1);

        if (!f.collected) {
            const dTop = dino.ducking ? dino.y + 30 : dino.y;
            const dBot = dino.y + dino.height;
            const dRight = dino.x + dino.width;

            if (dRight > f.x && dino.x < f.x + f.width && dBot > f.y && dTop < f.y + f.height) {
                f.collected = true;
                score += 100; // ¡Reducido a 100 puntos!
                sfx.flag();
                flags.splice(index, 1);
            }
        }
    });
}

function drawScore() {
    ctx.fillStyle = currentScenario === 'night' ? '#fff' : '#000'; 
    ctx.font = 'bold 28px Courier New'; 
    ctx.textAlign = 'right';
    ctx.shadowBlur = 5; 
    ctx.shadowColor = currentScenario === 'night' ? '#fff' : '#fff';
    ctx.fillText('PUNTOS: ' + String(Math.floor(score)).padStart(5, '0'), canvas.width - 20, 40);
    ctx.shadowBlur = 0;
}

// --- LOOP PRINCIPAL ---
function gameLoop() {
    if (!isPlaying) return;

    ctx.clearRect(0, 0, canvas.width, canvas.height);
    drawBackground();
    ground.draw();
    drawFlags();
    dino.draw(); 
    dino.update(); 
    drawObstacles(); 
    updateObstacles();
    updateFlags();
    drawScore();

    score += 0.1;

    // Sonido cada 100 puntos
    if (Math.floor(score) > lastScoreSound && Math.floor(score) % 100 === 0) {
        sfx.score();
        lastScoreSound = Math.floor(score);
        if (gameSpeed < 15) gameSpeed += 0.2;
    }

    // Sonido especial cada 1000 puntos
    if (Math.floor(score) > lastMilestoneSound && Math.floor(score) % 1000 === 0) {
        sfx.milestone();
        lastMilestoneSound = Math.floor(score);
    }

    animationId = requestAnimationFrame(gameLoop);
}

function startGame() {
    if (audioCtx.state === 'suspended') audioCtx.resume();
    overlay.style.display = 'none'; 
    isPlaying = true; 
    gameSpeed = 6; 
    score = 0; 
    obstacles = []; 
    flags = [];
    lastObstacleScore = 0;
    lastScoreSound = 0;
    lastMilestoneSound = 0;
    lastFlagScore = 0;
    currentScenario = 'night';
    scenarioMessageTimer = 0;
    dino.y = dino.groundY; 
    dino.jumping = false; 
    dino.ducking = false;
    if (musicOn) startMusic();
    gameLoop();
}

function gameOver() {

    sfx.die();

}

document.addEventListener('keydown', (e) => {
    if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(e.key)) {
        e.preventDefault(); 
    }

    if (e.key === 'Enter' || e.keyCode === 66 || e.keyCode === 23 || e.keyCode === 13) {
        const now = Date.now();
        if (now - lastOkPress < 300) {
            if (isPlaying && !dino.ducking) {
                dino.duck(true);
                if (duckTimer) clearTimeout(duckTimer);
                duckTimer = setTimeout(() => dino.duck(false), 800);
            }
            lastOkPress = 0;
        } else {
            lastOkPress = now;
            if (!isPlaying) startGame(); 
            else dino.jump();
        }
    }

    if (e.key === 'ArrowUp') {
        if (!isPlaying) startGame(); 
        else dino.jump();
    }
});
