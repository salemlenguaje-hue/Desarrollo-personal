#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Firma y verificación Ed25519 para recibos de SALEM Neurociencia.

La clave pública debe obtenerse por un canal de confianza independiente.
La firma autentica el contenido firmado; no demuestra que los datos sean
verdaderos ni acredita por sí sola la fecha de creación.
"""

import argparse
import base64
import hashlib
import json
import os
import subprocess
import sys
import tempfile
from pathlib import Path

import trazas_salem as ts

ESQUEMA_FIRMA = "SALEM-ED25519-1"


def ejecutar_openssl(argumentos):
    """Ejecuta OpenSSL y explica los errores en español."""
    try:
        return subprocess.run(
            ["openssl", *argumentos],
            check=True,
            capture_output=True,
            text=True,
        )
    except FileNotFoundError as exc:
        raise RuntimeError("No se encontró OpenSSL en el sistema.") from exc
    except subprocess.CalledProcessError as exc:
        detalle = (exc.stderr or exc.stdout or "").strip()
        raise RuntimeError(
            "Falló OpenSSL" + (": " + detalle if detalle else ".")
        ) from exc


def escribir_nuevo(ruta, contenido, permisos=None):
    """Crea un archivo nuevo sin sobrescribir uno existente."""
    ruta = Path(ruta)
    ruta.parent.mkdir(parents=True, exist_ok=True)

    # El modo exclusivo impide sobrescribir archivos existentes.
    descriptor = os.open(
        str(ruta),
        os.O_WRONLY | os.O_CREAT | os.O_EXCL,
        permisos if permisos is not None else 0o644,
    )
    try:
        with os.fdopen(descriptor, "wb") as archivo:
            archivo.write(contenido)
            archivo.flush()
            os.fsync(archivo.fileno())
    except Exception:
        # Si falla la escritura, se elimina el archivo incompleto.
        try:
            ruta.unlink()
        except OSError:
            pass
        raise


def generar_claves(clave_privada, clave_publica):
    """Genera un par Ed25519 y protege la clave privada."""
    privada = Path(clave_privada)
    publica = Path(clave_publica)

    if privada.resolve() == publica.resolve():
        raise ValueError("Las rutas de las claves deben ser diferentes.")
    if privada.exists() or publica.exists():
        raise FileExistsError(
            "Una de las rutas ya existe. No se sobrescribió ningún archivo."
        )

    # Las claves se generan primero en una carpeta temporal.
    with tempfile.TemporaryDirectory(prefix="salem_claves_") as temporal:
        temporal = Path(temporal)
        priv_tmp = temporal / "privada.pem"
        pub_tmp = temporal / "publica.pem"

        ejecutar_openssl([
            "genpkey", "-algorithm", "ED25519",
            "-out", str(priv_tmp),
        ])
        ejecutar_openssl([
            "pkey", "-in", str(priv_tmp),
            "-pubout", "-out", str(pub_tmp),
        ])

        # La clave privada se crea con permisos restrictivos.
        escribir_nuevo(privada, priv_tmp.read_bytes(), 0o600)
        try:
            escribir_nuevo(publica, pub_tmp.read_bytes(), 0o644)
        except Exception:
            # Evita dejar una clave privada huérfana si falla la pública.
            privada.unlink(missing_ok=True)
            raise

    # Refuerza los permisos en sistemas que admiten chmod.
    try:
        privada.chmod(0o600)
    except OSError:
        print("ADVERTENCIA: no se pudieron reforzar los permisos de la clave.")

    print("Par de claves Ed25519 generado.")
    print("Clave privada:", privada)
    print("Clave pública:", publica)
    print("Protegé la clave privada y guardá una copia confiable de la pública.")


def cargar_recibo(ruta):
    """Lee un recibo JSON y prepara su representación canónica."""
    try:
        objeto = json.loads(Path(ruta).read_text(encoding="utf-8"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError(f"No se pudo leer el recibo: {exc}") from exc

    if not isinstance(objeto, dict):
        raise ValueError("El recibo debe ser un objeto JSON.")

    # Se reutiliza la canonicalización documentada por el proyecto.
    contenido = ts.canonical(objeto).encode("utf-8")
    return contenido


def firmar_recibo(ruta_recibo, ruta_privada, ruta_firma):
    """Firma el contenido canónico del recibo y crea una firma separada."""
    contenido = cargar_recibo(ruta_recibo)
    privada = Path(ruta_privada)

    if not privada.is_file():
        raise FileNotFoundError("No se encontró la clave privada indicada.")

    with tempfile.TemporaryDirectory(prefix="salem_firma_") as temporal:
        temporal = Path(temporal)
        mensaje = temporal / "recibo.canonico"
        firma = temporal / "firma.bin"
        mensaje.write_bytes(contenido)

        ejecutar_openssl([
            "pkeyutl", "-sign", "-rawin",
            "-inkey", str(privada),
            "-in", str(mensaje),
            "-out", str(firma),
        ])

        # El archivo lateral no contiene la clave privada.
        documento = {
            "signature_schema": ESQUEMA_FIRMA,
            "algorithm": "Ed25519",
            "signed_anchor_sha256": hashlib.sha256(contenido).hexdigest(),
            "signature_base64": base64.b64encode(
                firma.read_bytes()
            ).decode("ascii"),
        }

        salida = (
            json.dumps(
                documento, ensure_ascii=False, sort_keys=True,
                indent=2, allow_nan=False
            ) + "\n"
        ).encode("utf-8")

        escribir_nuevo(ruta_firma, salida)

    print("Firma Ed25519 creada:", ruta_firma)
    print("SHA-256 del contenido firmado:",
          hashlib.sha256(contenido).hexdigest())
    print("Guardá la firma separada del recibo y protegé la clave privada.")


def verificar_firma(ruta_recibo, ruta_publica, ruta_firma):
    """Verifica la firma usando una clave pública indicada explícitamente."""
    contenido = cargar_recibo(ruta_recibo)

    try:
        documento = json.loads(
            Path(ruta_firma).read_text(encoding="utf-8")
        )
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError(f"No se pudo leer el archivo de firma: {exc}") from exc

    if not isinstance(documento, dict):
        raise ValueError("El archivo de firma debe ser un objeto JSON.")
    if documento.get("signature_schema") != ESQUEMA_FIRMA:
        raise ValueError("Esquema de firma desconocido.")
    if documento.get("algorithm") != "Ed25519":
        raise ValueError("El algoritmo de firma no es Ed25519.")

    resumen = hashlib.sha256(contenido).hexdigest()
    if documento.get("signed_anchor_sha256") != resumen:
        print("VERIFICACIÓN: FALLO")
        print("El contenido del recibo no coincide con el contenido firmado.")
        return False

    try:
        firma_binaria = base64.b64decode(
            documento["signature_base64"], validate=True
        )
    except (KeyError, ValueError, TypeError) as exc:
        raise ValueError("La firma Base64 falta o no es válida.") from exc

    publica = Path(ruta_publica)
    if not publica.is_file():
        raise FileNotFoundError("No se encontró la clave pública indicada.")

    with tempfile.TemporaryDirectory(prefix="salem_verificar_") as temporal:
        temporal = Path(temporal)
        mensaje = temporal / "recibo.canonico"
        firma = temporal / "firma.bin"
        mensaje.write_bytes(contenido)
        firma.write_bytes(firma_binaria)

        try:
            ejecutar_openssl([
                "pkeyutl", "-verify", "-rawin",
                "-pubin", "-inkey", str(publica),
                "-in", str(mensaje),
                "-sigfile", str(firma),
            ])
        except RuntimeError:
            print("VERIFICACIÓN: FALLO")
            print("La firma no es válida para esta clave pública y este recibo.")
            return False

    print("VERIFICACIÓN: OK")
    print("La firma corresponde al contenido y a la clave pública indicada.")
    print("SHA-256 del contenido:", resumen)
    print("La confianza en la clave pública debe comprobarse por separado.")
    return True


def main():
    """Interfaz de comandos del módulo de firma."""
    parser = argparse.ArgumentParser(
        description="SALEM Neurociencia: firma Ed25519 de recibos"
    )
    sub = parser.add_subparsers(dest="comando", required=True)

    claves = sub.add_parser("generar-claves", help="Crear un par de claves")
    claves.add_argument("--privada", required=True, help="Ruta de la clave privada")
    claves.add_argument("--publica", required=True, help="Ruta de la clave pública")

    firmar = sub.add_parser("firmar", help="Firmar un recibo existente")
    firmar.add_argument("--recibo", required=True)
    firmar.add_argument("--privada", required=True)
    firmar.add_argument("--salida", required=True, help="Archivo de firma nuevo")

    verificar = sub.add_parser("verificar", help="Verificar una firma existente")
    verificar.add_argument("--recibo", required=True)
    verificar.add_argument("--publica", required=True)
    verificar.add_argument("--firma", required=True)

    args = parser.parse_args()

    try:
        if args.comando == "generar-claves":
            generar_claves(args.privada, args.publica)
        elif args.comando == "firmar":
            firmar_recibo(args.recibo, args.privada, args.salida)
        elif args.comando == "verificar":
            resultado = verificar_firma(
                args.recibo, args.publica, args.firma
            )
            sys.exit(0 if resultado else 2)
    except (OSError, ValueError, TypeError, KeyError, RuntimeError) as exc:
        print("ERROR:", exc, file=sys.stderr)
        sys.exit(1)


if __name__ == "__main__":
    main()

def normalize_strings(obj):
    """Normaliza recursivamente valores y claves de texto a Unicode NFC."""
    if isinstance(obj, str):
        return unicodedata.normalize("NFC", obj)
    if isinstance(obj, list):
        return [normalize_strings(x) for x in obj]
    if isinstance(obj, dict):
        out = {}
        for key, value in obj.items():
            if not isinstance(key, str):
                raise TypeError("Las claves de objetos JSON deben ser texto.")
            nk = unicodedata.normalize("NFC", key)
            if nk in out:
                raise ValueError(f"Colisión de claves después de normalizar NFC: {nk!r}")
            out[nk] = normalize_strings(value)
        return out
    if obj is None or isinstance(obj, (bool, int, float)):
        return obj
    raise TypeError(f"Tipo no serializable admitido: {type(obj).__name__}")

Esto calcula un hash del contenido del recibo, pero no utiliza una clave secreta ni una firma digital.
Por lo tanto:
Si se altera un evento y se conserva el recibo original, la verificación debería detectar la discrepancia.
Si se altera toda la cadena y se conserva el recibo original, también debería detectar la discrepancia.

El anclaje es una referencia de integridad, pero no es, por sí mismo, una prueba de autenticidad independiente.

Pero el recibo actual no tiene una identidad autenticada de manera independiente. Su raíz de integridad es otro hash sin firma ni secreto. Por eso, quien puede reemplazar tanto la cadena como el recibo puede reconstruir un conjunto consistente.

#!/usr/bin/env python3
"""SALEM Neurociencia V2: trazabilidad 0.2. Biblioteca estándar de Python."""
import argparse
import datetime as dt
import hashlib
import json
import os
import sys
import unicodedata
from pathlib import Path

SCHEMA = "1.1"
ANCHOR_SCHEMA = "SALEM-ANCHOR-1"
MANIFEST = "manifest.json"
EVENTS = "events.jsonl"
GENESIS = "GENESIS"


def now():
    return dt.datetime.now(dt.timezone.utc).isoformat(timespec="milliseconds").replace("+00:00", "Z")


def normalize_strings(obj):
    """Normaliza recursivamente valores y claves de texto a Unicode NFC."""
    if isinstance(obj, str):
        return unicodedata.normalize("NFC", obj)
    if isinstance(obj, list):
        return [normalize_strings(x) for x in obj]
    if isinstance(obj, dict):
        out = {}
        for key, value in obj.items():
            if not isinstance(key, str):
                raise TypeError("Las claves de objetos JSON deben ser texto.")
            nk = unicodedata.normalize("NFC", key)
            if nk in out:
                raise ValueError(f"Colisión de claves después de normalizar NFC: {nk!r}")
            out[nk] = normalize_strings(value)
        return out
    if obj is None or isinstance(obj, (bool, int, float)):
        return obj
    raise TypeError(f"Tipo no serializable admitido: {type(obj).__name__}")


def canonical(obj):
    """Serialización determinista dentro de esta implementación Python; no declara JCS/RFC 8785."""
    return json.dumps(normalize_strings(obj), ensure_ascii=False, sort_keys=True,
                      separators=(",", ":"), allow_nan=False)


def sha_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha(text):
    return sha_bytes(text.encode("utf-8"))


def fsync_directory(directory):
    """Mejor esfuerzo: algunos sistemas/plataformas no permiten fsync de directorios."""
    flags = getattr(os, "O_RDONLY", 0) | getattr(os, "O_DIRECTORY", 0)
    fd = None
    try:
        fd = os.open(str(directory), flags)
        os.fsync(fd)
        return True
    except (OSError, AttributeError):
        return False
    finally:
        if fd is not None:
            os.close(fd)


def durable_write(path, text, append=False):
    path.parent.mkdir(parents=True, exist_ok=True)
    mode = "a" if append else "w"
    with path.open(mode, encoding="utf-8", newline="\n") as f:
        f.write(text)
        f.flush()
        os.fsync(f.fileno())
    fsync_directory(path.parent)


def atomic_write(path, text):
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    try:
        with tmp.open("w", encoding="utf-8", newline="\n") as f:
            f.write(text)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, path)
        fsync_directory(path.parent)
    finally:
        if tmp.exists():
            try:
                tmp.unlink()
            except OSError:
                pass


def load_manifest(d):
    p = d / MANIFEST
    if not p.exists():
        raise FileNotFoundError(f"No existe {p}; ejecutá iniciar primero.")
    return json.loads(p.read_text(encoding="utf-8"))


def write_manifest(d, obj):
    atomic_write(d / MANIFEST, json.dumps(normalize_strings(obj), ensure_ascii=False, indent=2, allow_nan=False) + "\n")


def read_events(path):
    events, issues = [], []
    if not path.exists():
        return events, [f"No existe {path.name}."]
    try:
        lines = path.read_text(encoding="utf-8").splitlines()
    except (OSError, UnicodeError) as exc:
        return [], [f"No se pudo leer {path.name}: {exc}"]
    for n, line in enumerate(lines, 1):
        if not line.strip():
            issues.append(f"Línea {n}: línea vacía inesperada.")
            continue
        try:
            e = json.loads(line)
            if not isinstance(e, dict):
                raise ValueError("el evento no es un objeto")
            e = normalize_strings(e)
            e["_line"] = n
            events.append(e)
        except (json.JSONDecodeError, ValueError, TypeError) as exc:
            issues.append(f"Línea {n}: registro inválido: {exc}")
    return events, issues


def append_event(d, event_type, payload):
    m = load_manifest(d)
    path = d / m.get("events_file", EVENTS)
    events, issues = read_events(path)
    if issues:
        raise RuntimeError("No se agrega a un registro con errores: " + "; ".join(issues))
    # No se permite continuar silenciosamente después de un evento de cierre.
    if events and events[-1].get("event_type") == "run_completed":
        raise RuntimeError("La corrida está cerrada con run_completed; no se agregan más eventos.")
    payload = normalize_strings(payload)
    prev = events[-1].get("event_hash", GENESIS) if events else GENESIS
    event = {
        "schema_version": SCHEMA, "experiment_id": m["experiment_id"],
        "run_id": m["run_id"], "sequence": len(events) + 1,
        "timestamp_utc": now(), "event_type": normalize_strings(event_type),
        "payload": payload, "previous_event_hash": prev
    }
    event["event_hash"] = sha(canonical(event))
    durable_write(path, canonical(event) + "\n", append=True)
    return event


def start(d, experiment_id, architecture, task, seed, code_version):
    d.mkdir(parents=True, exist_ok=True)
    if (d / MANIFEST).exists() or (d / EVENTS).exists():
        raise FileExistsError("El directorio ya contiene archivos de ejecución; no se sobrescribe.")
    m = {
        "schema_version": SCHEMA, "experiment_id": normalize_strings(experiment_id), "run_id": d.name,
        "created_at_utc": now(), "architecture_id": normalize_strings(architecture), "task_id": normalize_strings(task),
        "seed": seed, "code_version": normalize_strings(code_version), "hash_algorithm": "SHA-256",
        "event_serialization": "Python JSON sorted keys + compact separators; NFC strings; UTF-8",
        "unicode_normalization": "NFC", "events_file": EVENTS, "status": "initialized",
        "durability": "file fsync + best-effort parent-directory fsync; platform dependent"
    }
    write_manifest(d, m)
    durable_write(d / EVENTS, "")
    append_event(d, "run_started", {
        "architecture_id": architecture, "task_id": task,
        "seed": seed, "code_version": code_version
    })
    return m


def audit(d):
    m = load_manifest(d)
    events, issues = read_events(d / m.get("events_file", EVENTS))
    prev, seq = GENESIS, 1
    for e in events:
        line = e.pop("_line")
        if e.get("schema_version") != SCHEMA:
            issues.append(f"Línea {line}: schema_version inesperada.")
        if e.get("sequence") != seq:
            issues.append(f"Línea {line}: secuencia esperada {seq}, hallada {e.get('sequence')}.")
        if e.get("previous_event_hash") != prev:
            issues.append(f"Línea {line}: enlace al evento anterior incorrecto.")
        stored = e.get("event_hash")
        body = dict(e)
        body.pop("event_hash", None)
        try:
            calculated = sha(canonical(body))
        except (TypeError, ValueError) as exc:
            calculated = None
            issues.append(f"Línea {line}: no se pudo calcular hash: {exc}")
        if stored != calculated:
            issues.append(f"Línea {line}: event_hash no coincide; registro modificado o corrupto.")
        if e.get("experiment_id") != m.get("experiment_id"):
            issues.append(f"Línea {line}: experiment_id no coincide con el manifiesto.")
        if e.get("run_id") != m.get("run_id"):
            issues.append(f"Línea {line}: run_id no coincide con el manifiesto.")
        prev, seq = stored, seq + 1
    if not events:
        issues.append("No hay eventos registrados.")
    return {"ok": not issues, "event_count": len(events),
            "last_event_hash": events[-1].get("event_hash") if events else None,
            "issues": issues}


def anchor_core(d, audit_result):
    manifest_bytes = (d / MANIFEST).read_bytes()
    return {
        "anchor_schema": ANCHOR_SCHEMA,
        "run_id": load_manifest(d)["run_id"],
        "manifest_sha256": sha_bytes(manifest_bytes),
        "events_file": load_manifest(d).get("events_file", EVENTS),
        "event_count": audit_result["event_count"],
        "last_event_hash": audit_result["last_event_hash"],
        "anchored_at_utc": now()
    }


def create_anchor(d, anchor_path):
    result = audit(d)
    if not result["ok"]:
        raise RuntimeError("No se sella una corrida cuya auditoría interna falla: " + "; ".join(result["issues"]))
    core = anchor_core(d, result)
    core["integrity_root"] = sha(canonical(core))
    atomic_write(anchor_path, json.dumps(core, ensure_ascii=False, indent=2, allow_nan=False) + "\n")
    return core


def verify_anchor(d, anchor_path):
    issues = []
    try:
        anchor = json.loads(anchor_path.read_text(encoding="utf-8"))
        if not isinstance(anchor, dict):
            raise ValueError("el anclaje no es un objeto JSON")
    except (OSError, UnicodeError, json.JSONDecodeError, ValueError) as exc:
        return {"ok": False, "issues": [f"No se pudo leer el anclaje: {exc}"]}
    stored_root = anchor.get("integrity_root")
    core = dict(anchor)
    core.pop("integrity_root", None)
    if stored_root != sha(canonical(core)):
        issues.append("El integrity_root del anclaje no coincide; el recibo fue alterado o está corrupto.")
    internal = audit(d)
    if not internal["ok"]:
        issues.append("La auditoría interna falla: " + "; ".join(internal["issues"]))
    try:
        current_manifest_hash = sha_bytes((d / MANIFEST).read_bytes())
        m = load_manifest(d)
        if anchor.get("run_id") != m.get("run_id"):
            issues.append("run_id no coincide con el anclaje.")
        if anchor.get("manifest_sha256") != current_manifest_hash:
            issues.append("manifest_sha256 no coincide con el anclaje.")
        if anchor.get("events_file") != m.get("events_file", EVENTS):
            issues.append("events_file no coincide con el anclaje.")
        if anchor.get("event_count") != internal["event_count"]:
            issues.append("event_count no coincide con el anclaje.")
        if anchor.get("last_event_hash") != internal["last_event_hash"]:
            issues.append("last_event_hash no coincide con el anclaje.")
    except (OSError, ValueError, KeyError) as exc:
        issues.append(f"No se pudo comparar el anclaje: {exc}")
    return {"ok": not issues, "issues": issues, "internal_audit_ok": internal["ok"]}
